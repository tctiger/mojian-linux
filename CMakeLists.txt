cmake_minimum_required(VERSION 3.20)
project(MojianLinux VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets PrintSupport Network Svg DBus Xml Concurrent)
# Compile the same pinned, patched C parsers as the macOS edition.
file(GLOB COMMON_C CONFIGURE_DEPENDS Vendor/cmark/src/*.c)
file(GLOB GFM_C CONFIGURE_DEPENDS Vendor/cmark-gfm/src/*.c Vendor/cmark-gfm/extensions/*.c)
list(FILTER COMMON_C EXCLUDE REGEX "/main\\.c$")
list(FILTER GFM_C EXCLUDE REGEX "/main\\.c$")
add_library(mojian_common OBJECT ${COMMON_C})
target_include_directories(mojian_common PRIVATE Vendor/Generated/Common Vendor/cmark/src)
target_compile_definitions(mojian_common PRIVATE CMARK_STATIC_DEFINE)
target_compile_options(mojian_common PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/Vendor/Generated/Common/prefix.h)
add_library(mojian_markdown STATIC ${GFM_C} Vendor/MojianMarkdown/MojianMarkdown.c Vendor/MojianMarkdown/MojianInline.c $<TARGET_OBJECTS:mojian_common>)
target_include_directories(mojian_markdown PUBLIC Vendor/MojianMarkdown PRIVATE Vendor/Generated Vendor/cmark-gfm/src Vendor/cmark-gfm/extensions)
target_compile_definitions(mojian_markdown PRIVATE CMARK_GFM_STATIC_DEFINE CMARK_GFM_EXTENSIONS_STATIC_DEFINE)
target_link_libraries(mojian_markdown PUBLIC pthread)
add_library(mojian_core STATIC src/editor.cpp src/editor.h src/markdown.cpp src/markdown.h src/media.cpp src/media.h src/services.cpp src/services.h src/live.cpp src/live.h src/spell.cpp src/spell.h resources.qrc)
string(TIMESTAMP MOJIAN_BUILD_TIME "%Y-%m-%d %H:%M:%S UTC" UTC)
target_compile_definitions(mojian_core PUBLIC MOJIAN_BUILD_TIME="${MOJIAN_BUILD_TIME}")
target_link_libraries(mojian_core PUBLIC Qt6::Widgets Qt6::PrintSupport Qt6::Network Qt6::Svg Qt6::DBus Qt6::Xml Qt6::Concurrent mojian_markdown)
target_include_directories(mojian_core PUBLIC src)
add_executable(mojian src/main.cpp)
target_link_libraries(mojian PRIVATE mojian_core)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(mojian_tests tests/editor_test.cpp)
    target_link_libraries(mojian_tests PRIVATE mojian_core Qt6::Test)
    add_executable(mojian_parity_tests tests/parity_test.cpp)
    target_link_libraries(mojian_parity_tests PRIVATE mojian_core Qt6::Test)
    target_compile_definitions(mojian_parity_tests PRIVATE MOJIAN_SPEC_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/Specs")
    add_test(NAME parity COMMAND mojian_parity_tests)
    set_tests_properties(parity PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME native-editor COMMAND mojian_tests)
    set_tests_properties(native-editor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
include(GNUInstallDirs)
install(TARGETS mojian RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES resources/io.mojian.Editor.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/mojian.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME mojian.png)
install(FILES resources/help-zh-CN.md resources/ThirdPartyNotices.txt README.md PARITY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/mojian)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/mojian RENAME LICENSE)
install(FILES resources/ThirdPartyNotices.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/mojian)
install(FILES NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/mojian)
install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/mojian)
